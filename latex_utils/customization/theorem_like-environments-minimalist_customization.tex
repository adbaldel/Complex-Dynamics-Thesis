\usepackage[most]{tcolorbox}

% --- Define the base style mapping your exact mdframed geometry ---
\tcbset{
    theoremLineStyle/.style={
        blanker,                  % Removes background and all 4 borders
        breakable,
        left=12pt, right=0pt, top=8pt, bottom=8pt,
        before skip=10pt, after skip=10pt,
        borderline west={2pt}{0pt}{#1} % Width=3pt, Offset=0pt, Color=Argument
    }
}
