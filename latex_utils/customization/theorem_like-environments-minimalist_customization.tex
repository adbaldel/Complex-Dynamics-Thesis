\usepackage{mdframed}

% --- Define Base Frame Style --------------------------------------------------
\mdfdefinestyle{theoremLineStyle}{
    topline=false,
    bottomline=false,
    rightline=false,
    linewidth=2pt,          % Width of the left line
    innertopmargin=-4pt,     % Padding inside the frame
    innerbottommargin=2pt,
    innerleftmargin=10pt,
    innerrightmargin=0pt,
    skipabove=4pt,         % Space above the environment
    skipbelow=4pt          % Space below the environment
}
