\usepackage[most]{tcolorbox}

% --- Define the base style mapping your exact mdframed geometry ---
\tcbset{
    theoremLineStyle/.style={
        breakable,
        left=10pt, right=10pt, top=8pt, bottom=8pt,
        before skip=10pt, after skip=10pt,
        colframe=#1,              % Border color becomes your solid pastel color
        colback=#1!15,            % Even lighter background fill (15% color)
        arc=4pt,                  % Radius for smoothly rounded corners
        boxrule=1pt               % Thickness of the full border frame
    }
}
