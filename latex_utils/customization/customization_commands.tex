\usepackage{soul}

% --- Command for highlighting in a color --------------------------------------
\newcommand{\customhighlight}[2]{
    \sethlcolor{#1}
    \hl{#2}
} % Example: \customhighlight{pastelGreen}{your text here}

\newcommand{\customunderline}[2]{
    \setulcolor{#1}
    \ul{#2}
} % Example: \customunderline{pastelGreen}{your text here}
