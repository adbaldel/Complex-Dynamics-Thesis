\usepackage{mdframed}

% --- Define Base Frame Style --------------------------------------------------
\mdfdefinestyle{theoremLineStyle}{
    topline=false,
    bottomline=false,
    rightline=false,
    linewidth=3pt,          % Width of the left line
    innertopmargin=2pt,     % Padding inside the frame
    innerbottommargin=4pt,
    innerleftmargin=10pt,
    innerrightmargin=0pt,
    skipabove=10pt,         % Space above the environment
    skipbelow=10pt          % Space below the environment
}

% --- Register Environments with mdframed to add the left lines ----------------
