\chapter*{Notation}
\addcontentsline{toc}{chapter}{Notation}

{
\centering

\noindent {\large Basic Math}
\vspace{1em}
\begin{itemize}
    \item $fg$: Composition of $f$ and $g$ ($f \circ g$).
    \item $f^n$: The composition of $f$ with itself $n$ times.
    \item $\Id = f^0$: Identity function.
    \item $f^{(n)}$: $n$-th derivative of $f$.
    \item $\sim$: Equivalence relation.
    \item $[x]$: Equivalence class of $x$ (in some contexts orbit of $x$).
    \item $\N$: Naturals without $0$.
    \item $\Nzero$: Naturals with $0$.
    \item $q \mid n$: $q$ divides $n$.
    \item $q \nmid n$: $q$ does not divide $n$.
\end{itemize}

\vspace{2em}
\noindent {\large Complex Analysis}
\vspace{1em}
\begin{itemize}
    \item $\ExtC$: Extended complex plane.
    \item $\Delta$: Unit disk in $\C$.
    \item $R(z)$: Rational function in complex variable $z$.
    \item $\sigma$: Chordal metric.
    \item $\sigma_0$: Spheric metric.
    \item $\gamma$: A curve.
    \item $E(\gamma)$: Exterior component of a closed curve.
    \item $v_f(z)$: Valency of $f$ at $z$.
    \item $\delta_R(z)$: Deficiency of $R$ at $z$.
    \item $\automorphism{D}$: Group of automorphisms of $D$.
\end{itemize}

\vspace{2em}
\noindent {\large Topology}
\vspace{1em}
\begin{itemize}    
    \item $N$: Neighborhood of a point.
    \item $\diameter(E)$: Diameter of $E$.
    \item $\interior{E}$: Interior of $E$.
    \item $\closure{E}$: Closure of $E$.
    \item $\boundary{E}$: Boundary of $E$.
    \item $\derived{E}$: Derived set of $E$.
    \item $\chi(S)$: Euler's characteristic of $S$.
\end{itemize}

\vspace{2em}
\noindent {\large Complex Dynamics}
\vspace{1em}
\begin{itemize}
    \item $\zeta$: Fixed, periodic or pre-periodic point.
    \item $\deg(R)$: Degree of $R$.
    \item $m(R, z_0)$: Multiplier of $R$ at $z_0$.
    \item $F$ or $F(R)$: Fatou set (of $R$).
    \item $J$ or $J(R)$: Julia set (of $R$).
    \item $C$ or $C(R)$: Set of critical points (of $R$).
    \item $V$ or $V(R)$: Set of critical values (of $R$).
    \item $E(R)$: Set of exceptional points of $R$.
    \item $\orbit(z)$ or $[z]$: Orbit of $z$.
    \item $\orbit^-(z)$: Backward orbit of $z$.
    \item $\orbit^+(z)$: Forward orbit of $z$.
    \item $\mathbb{F}(F_0)$: The class of limit functions on $F_0$.
\end{itemize}
}
